cmake_minimum_required(VERSION 3.16)
project(AacFrameParser LANGUAGES CXX)

add_executable(aac_frame_parser main.cpp adts_parser.cpp)
target_compile_features(aac_frame_parser PRIVATE cxx_std_17)
target_include_directories(aac_frame_parser PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    target_compile_options(aac_frame_parser PRIVATE /W4 /permissive-)
else()
    target_compile_options(aac_frame_parser PRIVATE -Wall -Wextra -Wpedantic)
endif()
